cmake_minimum_required(VERSION 3.12)
project(FoodpandaOrderCalculator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(order_core src/restaurant.cpp src/menu.cpp src/quantity.cpp src/delivery.cpp src/payment.cpp src/validation.cpp)
target_include_directories(order_core PUBLIC include)
add_executable(foodpanda src/main.cpp)
target_link_libraries(foodpanda PRIVATE order_core)
enable_testing()
add_executable(test_calculations tests/test_calculations.cpp)
target_link_libraries(test_calculations PRIVATE order_core)
add_test(NAME calculations COMMAND test_calculations)
